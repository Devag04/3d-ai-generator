// 2. Build Procedural Three.js Mesh Group
function buildModelFromBlueprint(blueprint) {
  const group = new THREE.Group();
  blueprint.elements.forEach(el => {
    let geom;
    if(el.type === 'box') geom = new THREE.BoxGeometry(...el.size);
    else if(el.type === 'cylinder') geom = new THREE.CylinderGeometry(...el.size);
    else if(el.type === 'sphere') geom = new THREE.SphereGeometry(...el.size);
    
    const mat = new THREE.MeshStandardMaterial({
      color: el.color,
      roughness: el.roughness,
      metalness: el.metalness
    });
    const mesh = new THREE.Mesh(geom, mat);
    mesh.position.set(...el.pos);
    mesh.rotation.set(...el.rot);
    group.add(mesh);
  });
  return group;
}

Deployment Guide (Vercel, Render, Hugging Face)

1. Deploying on Vercel

  1. Push project files to a GitHub repository containing index.html.
  2. Connect your GitHub account to Vercel.
  3. Select "Import Repository" -> Keep default Static HTML framework settings.
  4. Click Deploy. Instant HTTPS URL generated in under 30 seconds.

2. Deploying on Render

  1. Log into Render dashboard and click New + -> Static Site.
  2. Connect your repository and specify Publish Directory as ./.
  3. Click Create Static Site.

3. Deploying on Hugging Face Spaces

  1. Create a new Space on Hugging Face.
  2. Choose Static HTML as the Space SDK type.
  3. Upload index.html and commit to main branch.
